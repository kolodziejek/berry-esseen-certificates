\section{Fixed-target extremality and conservation}\label{v9:core:section}
Fix $n\ge2$ and $c>0$. For a standardized marginal with finite third
absolute moment, write
\begin{gather*}
 z=H/\sqrt n,\quad F=\Pp(S_n\le H),\quad
 a=\sqrt n\{F-\Phi(z)\}-c\rho,\\
 \varphi=\phi(z),\quad\beta=\E[X|X|],\quad\gamma=\E X^3.
\end{gather*}
The threshold $H$ is raw, and $a$ is not divided by $\rho$.

\begin{proposition}[Attainment and contact]\label{v9:core:attainment}
The supremum of $a$ over standardized marginals and thresholds is attained.
Every attaining pair has positive closed-CDF discrepancy and compact support.
On the whole line,
\begin{align}\label{v9:core:contact}
 I(y)={}&n^{3/2}\{h(y)-F\}+ny\varphi
 +\tfrac12H\varphi(y^2-1)\notag\\
 &+\tfrac32c\rho(y^2-1)-c\{|y|^3-\rho-3y\beta\}\le0,
 \qquad h(y)=\Pp(S_{n-1}\le H-y),
\end{align}
with equality on the entire closed support. No sign assumption on $a$ or
$H$ is needed.
\end{proposition}
\begin{proof}
The supremum $A$ satisfies $-c<A\le\sqrt n-c$: the lower inequality uses
symmetric Bernoulli at $H=n$. A maximizing sequence therefore has bounded
$\rho_j$ and a positive discrepancy bounded away from zero.
Chebyshev bounds the negative threshold tail by $n/H_j^2$, and the
positive discrepancy is bounded by the normal tail. Hence $H_j$ is bounded.
Pass to $P_j\Rightarrow P$, $H_j\to H$, $\rho_j\to b$.
A common third-moment ceiling $M$ gives
\[
 \int_{|x|>K}x^2dP_j\le M/K,\qquad
 \int_{|x|>K}|x|dP_j\le M/K^2.
\]
Truncation preserves mean and variance, while lower semicontinuity gives
$1\le\rho(P)\le b$. The sum laws converge weakly; apply the closed-set
inequality at $H+t$ and let $t\downarrow0$ to obtain
$\limsup F_j(H_j)\le F_P(H)$, including atoms. Thus
$a(P,H)\ge A+c\{b-\rho(P)\}\ge A$. Maximality gives attainment and
$\rho(P)=b$; third-moment convergence has not been assumed. The discrepancy
is positive because $A+c\rho\ge A+c>0$.

Contaminate by $(1-\epsilon)P+\epsilon\delta_y$, then center and scale,
evaluating at the standardized image of the same raw event. The mean and
variance are $\epsilon y$ and $1+\epsilon(y^2-1)-\epsilon^2y^2$.
The derivatives at zero of the raw event probability, centered third
absolute moment and normal argument are, respectively,
\[
 n(h(y)-F),\qquad |y|^3-\rho-3y\beta,\qquad
 -\sqrt n y-H(y^2-1)/(2\sqrt n).
\]
The event probability is a polynomial even at collisions; dominated
moment differentiation uses $3|y|(|x|+|y|)^2$.
Differentiating the excess gives $I(y)\le0$ and $\int I\,dP=0$.
The closed-event function $h$, hence $I$, is upper semicontinuous.
Strict negativity at a support point would persist on a positive-mass
neighborhood, contradicting that integral. Finally
$I(y)\le-c|y|^3+O(y^2+|y|+1)$, so contact excludes both distant rays.
\end{proof}

\paragraph{Reflection and first failure.}\label{v9:core:reflection}
For every law, including an atom at $H$,
\[
 \Pp(-S_n\le-H)-\Phi(-z)
 =\Phi(z)-\Pp(S_n<H)\ge\Phi(z)-\Pp(S_n\le H).
\]
Thus $C_n$ is also the supremum of positive closed scores $r$.
Since $a=\rho(r-c)$, $C_n<c$ implies $A\le C_n-c<0$, and $C_n>c$
implies $A>0$. If $C_n=c$, choose $r_j\uparrow c$ with $r_j\ge c/2$;
then $\rho_j\le2\sqrt n/c$ gives $a_j\to0$, so $A=0$.
Consequently $C_n\ge c$ supplies an attained $a\ge0$ witness.
If a strict bound fails above a genuine weak base $C_j\le c$ for $j<N$,
the least failing index has this witness and true predecessors
$C_j\le c$ for all $j<n$.\label{v9:core:first-failure}

\subsection{An identity that retains contact residuals}
Write $\Delta=Q-h$, where contact is equivalent to $\Delta\ge0$ and
$\Delta=0$ on the closed support, and
\begin{equation}\label{v9:core:Q}
\begin{split}
 Q(y)={}&F+\frac{H\varphi+c\rho}{2n^{3/2}}
 -\frac{n\varphi+3c\beta}{n^{3/2}}y\\
 &-\frac{H\varphi+3c\rho}{2n^{3/2}}y^2
 +\frac{c|y|^3}{n^{3/2}}.
\end{split}
\end{equation}
For now allow any compact standardized law, without contact. Put
$\nu=\mathcal L(H-S_{n-1})$ and $\tau(t)=\E[X\ind_{X\ge t}]$.
Then $\tau\ge0$, $\int\tau=1$. On each bounded gap $(u,v)$ of the
closed support $K$, its interior value is a constant $\tau_{u,v}$.
Define
\begin{align*}
 J&=\sum\tau_{u,v}(v-u)^2,&
 J_1&=\sum\tau_{u,v}\int_u^v(t-u)t\,dt,\\
 J_2&=\sum\tau_{u,v}\int_u^v(t-u)t|t|\,dt,&
 L&=\sum\tau_{u,v}\int_{[u,v)}(t-u)\,d\nu(t).
\end{align*}
The residuals are
\begin{equation}\label{v9:core:residuals}
 \xi=\E[X\Delta(X)],\qquad
 B=\int_K\tau\Delta+\sum\tau_{u,v}(v-u)\Delta(v).
\end{equation}
The first integral is over Lebesgue measure on $K$ and cannot be omitted.
Compactness makes these quantities finite; the residual sum converges
absolutely since $\Delta$ is bounded and $\sum\tau_{u,v}(v-u)\le1$.
Let $M_4=\E X^4$, $M_5=\E|X|^5$ and set
\begin{gather*}
 K_q=M_5/4-\rho M_4/2-3\beta\gamma/2+\rho/2,
 \qquad K_1=\E[X|X|^3]-3\beta-3\rho\gamma/2,\\
 \Omega=J_1+(M_4-3)/6,\qquad q_z(x)=(x-z)|x-z|,\\
 \varepsilon=\E q_z(S_n/\sqrt n)-\E q_z(Z_0)
                 -2\gamma\varphi/(3\sqrt n),\qquad
 E_n=-n^{3/2}\varepsilon/2,
\end{gather*}
where $Z_0$ is standard normal. Define
\begin{equation}\label{v9:core:R}
 \mathcal R_c=c(3\beta J/2+3\rho J_1-3J_2-K_q)
 +H(cK_1+\varphi\Omega)-H^2(c\rho+\gamma\varphi/2)+E_n.
\end{equation}

\begin{theorem}[Conservation]\label{v9:core:general}
For every compact standardized law,
\begin{equation}\label{v9:core:offcontact}
 (n+H^2)a+nc\rho-\frac{n\varphi}{6}(\gamma+3J)+n^{3/2}L
 =\mathcal R_c+n^{3/2}(B-H\xi).
\end{equation}
At an attaining pair the residuals vanish, so
\begin{equation}\label{v9:core:conservation}
 (n+H^2)a+nc\rho-\frac{n\varphi}{6}(\gamma+3J)+n^{3/2}L=\mathcal R_c.
\end{equation}
\end{theorem}
\begin{proof}
Evaluate $\mathcal A=\int\tau\Delta$ twice. Inside a gap,
$h(t)=h(v)+\nu([t,v))$, whence Fubini gives
\[
 \int_u^v\Delta=(v-u)\Delta(v)
 -\int_u^v(t-u)Q'(t)dt-\int_{[u,v)}(t-u)d\nu(t).
\]
The atom at $v$ belongs to $h(v)$, not the last integral; the atom at $u$
has zero weight. Summing with the support integral yields
\begin{equation}\label{v9:core:first}
 \mathcal A=\frac{\varphi J}{2\sqrt n}
 +\frac{3c\beta J/2+(H\varphi+3c\rho)J_1-3cJ_2}{n^{3/2}}-L+B.
\end{equation}
For the other evaluation use signwise Fubini,
$\int\tau g'=\E[Xg(X)]$. In particular
$\int t\tau=\gamma/2$, $\int t^2\tau=M_4/3$ and
$\int|t|^3\tau=M_5/4$.
Condition on $S_{n-1}$ and take $g(x)=\min(S_{n-1}+x,H)$;
exchangeability gives
\[
 \int\tau h=\E[X_n\min(S_n,H)]
 =\frac1n\E[S_n\min(S_n,H)]
 =T:=\E[(Z^2-zZ)\ind_{Z\le z}],\qquad Z=S_n/\sqrt n.
\]
The endpoint integrand is zero, so atoms cause no extra term. Consequently
\begin{equation}\label{v9:core:second}
 \mathcal A=F-\frac{\gamma\varphi}{2\sqrt n}
 +\frac{H\varphi(3-M_4)/6+cK_q}{n^{3/2}}-T.
\end{equation}
Put $M=\E[Z\ind_{Z\le z}]$. Integrating $XQ(X)$ against $P$ gives
\begin{equation}\label{v9:core:firstmoment}
 M+\varphi=-\gamma H\varphi/(2n)+cK_1/n-\sqrt n\xi.
\end{equation}
The pointwise identity
$q_z(x)=(x-z)^2-2(x-z)^2\ind_{x\le z}$, also applied to $Z_0$, gives
\[
 T-\Phi(z)=-\varepsilon/2-\gamma\varphi/(3\sqrt n)
       +z(M+\varphi)-z^2(F-\Phi(z)).
\]
Insert this and \eqref{v9:core:firstmoment} into
\eqref{v9:core:second}, equate with \eqref{v9:core:first}, and use
$\sqrt n(F-\Phi(z))=c\rho+a$. Collecting terms proves
\eqref{v9:core:offcontact}, including the coefficient $n+H^2$.
Closed-support contact makes both residuals zero.
\end{proof}

\subsection{The resource and the order of its use}
The gap bounds proved in Lemma~\ref{ms:physical-defect} give
\begin{equation}\label{v9:core:defect}
 D=(3+\sqrt{10})\rho-\gamma-3J\ge0.
\end{equation}
At $c=\CE=(3+\sqrt{10})\phi(0)/6$, let
$w=H^2/(2n)$, $e=e^{-w}$. Conservation becomes
\begin{equation}\label{v9:core:positive}
 \mathcal R_c=(n+H^2)a+nc\rho(1-e)
              +\frac{n\phi(0)e}{6}D+n^{3/2}L.
\end{equation}
Every term is nonnegative when $a\ge0$.

\begin{corollary}[Raw ceiling and phase price]
\label{v9:core:license}\label{v9:core:phase-price}
At an attaining pair with $a\ge0$, suppose $n\ge N\ge2$, $\rho\ge l>0$, $0<c_-\le c$, and
$\mathcal R_c\le T<Nc_-l$. Negative $T$ excludes the pair. Otherwise
\begin{gather*}
 \gamma+3J>0,\qquad e\ge1-T/(Nc_-l)>0,\\
 \mathcal U:=(n+H^2)a+n\phi(0)D/6+n^{3/2}L
 \le\mathcal R_c-\tfrac12(c_-l-T/N)H^2\le\mathcal R_c.
\end{gather*}
\end{corollary}
\begin{proof}
Set $M_c=c\rho-\phi(0)D/6=\phi(0)(\gamma+3J)/6$.
The identity gives
$eM_c=(1+H^2/n)a+c\rho+\sqrt nL-\mathcal R_c/n
\ge c\rho-T/n>0$.
Since $D\ge0$, $M_c\le c\rho$, proving the sign and density claims.
Also $\mathcal R_c-\mathcal U=n(1-e)M_c$, and
\[
 n(1-e)M_c=n(e^w-1)eM_c
 \ge\tfrac12H^2(c\rho-T/n)
 \ge\tfrac12H^2(c_-l-T/N).
\]
Thus a raw ceiling licenses the density floor before the resource is
improved. The improved resource is not a new raw ceiling.
\end{proof}
