\section{The sign split and the physical gap defect}\label{ms:geometry-counts}
For a standardized law put $p=\Pp(X>0)$, $q=1-p$, $v=pq$ and
$m=\E X_+=\E X_->0$. Both sign classes have positive mass; any zero atom
belongs to the nonpositive class. With $B_0=\ind_{X>0}$, set
\begin{gather*}
 V=\E(X\mid B_0),\quad W=X-V,\quad \theta=\E W^2=1-m^2/v,\\
 \ell=m/v=\sqrt{(1-\theta)/v},\quad U=W/\ell.
\end{gather*}
The conditional means are $-m/q$ and $m/p$, so exactly
\begin{equation}\label{ms:bernoulli-coordinates}
 X=\ell(B_0+U-p),\qquad\E(U\mid B_0)=0,\quad
 s=\E U^2=\theta/\ell^2,\quad\lambda=ns,\quad v+s=\ell^{-2}.
\end{equation}
If $s=0$ the law is Bernoulli. Otherwise let $U_i$ be the conditional
noises and $w_i=\E U_i^2$, $\mu_i=\E U_i^3$. Expanding each actual sign
class, with conditional centering, gives
\begin{align}\label{ms:conditional-moments}
 \rho/\ell^3&=v(1-2v)+3v(w_0+w_1)-q\mu_0+p\mu_1,\notag\\
 \gamma/\ell^3&=v(q-p)+3v(w_1-w_0)+q\mu_0+p\mu_1,\notag\\
 \beta/\ell^2&=v(q-p)+pw_1-qw_0.
\end{align}
These are identities for the same marginal, not independently chosen
moment relaxations.

\begin{lemma}[Moment and gap bounds]\label{ms:physical-defect}
\label{ms:fourier:afcore:lem:physical-defect}
For $\sigma=\sqrt{10}$ and the actual $J$ of Section~\ref{v9:core:section},
\begin{gather}\label{ms:gap-family}
 \gamma^2+4m^2\le\rho^2,\quad J\le\rho+2m,\quad
 (2\rho)^{-1}\le m\le\tfrac12,\\
 |\beta|<1,\qquad
 \rho\ge\frac{1+\beta^2}{\sqrt{1-\beta^2}},\qquad
 D:=(3+\sigma)\rho-\gamma-3J\ge0.
 \label{ms:physical-defect-definition}
\end{gather}
If zero is in the closed support, $D\ge(\sigma-1)\rho$.
Otherwise its central gap $(-A,B)$ satisfies $A,B>0$, $AB\le1$ and
$J\le\rho+2mAB$.
\end{lemma}
\begin{proof}
Let $a_\pm=\E X_\pm^2=(1\pm\beta)/2$ and
$b_\pm=\E X_\pm^3=(\rho\pm\gamma)/2$.
Signwise Cauchy gives $a_\pm^2\le mb_\pm$ and
$m^2\le pa_+$, $m^2\le qa_-$. Hence
$m^2\le a_+a_-$ and $b_+b_-\ge m^2$, proving the circle inequality.
Also $m\rho\ge a_+^2+a_-^2=(1+\beta^2)/2$ and
$2m\le\sqrt{1-\beta^2}$ give the signed moment bound.
The bounds for $m$ follow from $\E|X|\le1$ and
$1\le\sqrt{\E|X|\E|X|^3}=\sqrt{2m\rho}$.

Signwise Tonelli gives
\begin{equation}\label{ms:tau-moments}
 \int\tau=1,\qquad 2\int|t|\tau(t)dt=\rho.
\end{equation}
Each gap on one side of zero obeys
$(b-a)^2\le2\int_a^b|t|dt$. A central gap has weight $m$ and excess
$2mAB$ over this integral; $0\le\E[(X+A)(X-B)]=1-AB$.
Summing proves the bounds for $J$, also for countably many gaps.
Cauchy in $\R^2$ gives $\gamma+6m\le\sqrt{10}\rho$, and therefore
$D=3(\rho+2m-J)+(\sigma\rho-\gamma-6m)\ge0$.
Without a central gap, $J\le\rho$ gives $D\ge(\sigma-1)\rho$.
\end{proof}

For the central gap define
\[
 T_+=\E X_+^3-mB^2,\quad T_-=\E X_-^3-mA^2,
 \quad R_+=\E X_+^2-mB,\quad R_-=\E X_-^2-mA.
\]
All four quantities are nonnegative. If $J_o=J-m(A+B)^2$,
Tonelli and the positive outer-gap inequality give
\[
 J_o^+\le2\int_B^\infty(t-B)\tau(t)dt
 =\E[X_+(X_+-B)^2]=T_+-2BR_+;
\]
the negative counterpart is $J_o^-\le T_--2AR_-$. Expanding $D$ yields
\begin{equation}\label{ms:defect-decomposition}
\begin{split}
 D={}&m\{\sqrt{\sigma+1}A-\sqrt{\sigma-1}B\}^2
       +(\sigma-1)T_++(\sigma+1)T_-\\
    &+3(T_++T_--J_o).
\end{split}
\end{equation}
Every term is nonnegative. Thus $T_++T_-\le D/2$ and, if $D<2$,
$m(A^2+B^2)=\rho-T_+-T_-\ge1-D/2$ gives $A+B\ge\sqrt{2-D}$.
The stronger outer-gap bounds also give the useful joint inequality
\[
 D\ge(\sigma-1)\E X_+^3+(\sigma+1)\E X_-^3
                         -6mAB+6BR_++6AR_-.
\]
This retains the actual gap and conditional noises before any costs are
weakened. It is the common starting point of the direct acquisition.
