\section{Elementary constants}\label{ms:normalization}\label{ms:gaussian-constant-proof}
\begin{lemma}\label{ms:gaussian-constants}
The standard normal density satisfies
\[
 \|\phi''\|_\infty=\phi(0)<2/5,\quad
 \|\phi'''\|_\infty\le2/\pi,\quad 1/(4\pi)<2/25,
\]
\[
 819/2000<\CE<20487/50000<41/100,\qquad
 \phi(0)>.39886>.3988=997/2500.
\]
Moreover $333/106<\pi<22/7$, $\sqrt\pi>4431/2500$, and
$0.4097321837023<\CE<0.4097321837025$.
\end{lemma}
\begin{proof}
For $J(t)=\int e^{-x^2/2}e^{itx}dx$, integration by parts gives $J'=-tJ$;
the squared Gaussian integral in polar coordinates gives $J(0)=\sqrt{2\pi}$.
Thus $\phi(x)=(2\pi)^{-1}\int e^{-t^2/2}e^{-itx}dt$.
Differentiating under integrable majorants gives
$\|\phi'''\|_\infty\le(2\pi)^{-1}\int|t|^3e^{-t^2/2}dt=2/\pi$;
likewise $\|\phi^{(4)}\|_\infty\le3\phi(0)$ and
$\|\phi^{(5)}\|_\infty\le8/\pi$.
The identity $\phi''=(x^2-1)\phi$ gives its exact norm:
the exterior maximum is $2e^{-3/2}\phi(0)<\phi(0)$.

Put $s_k(x)=\sum_{j=0}^{k-1}(-1)^jx^{2j+1}/(2j+1)$.
Tangent addition and the bounds
$0<4(1/5-1/(3\cdot5^3))-1/239
<4\arctan(1/5)-\arctan(1/239)<1<\pi/2$
fix the first-quadrant branch (use $\pi/2=2\int_0^1(1+t^2)^{-1}dt>1$).
Hence Machin's identity is $\pi=16\arctan(1/5)-4\arctan(1/239)$.
Integrating the finite geometric expansion of $(1+t^2)^{-1}$ gives
\[
 \pi_-:=16s_{12}(1/5)-4s_3(1/239)<\pi
 <16s_{11}(1/5)-4s_4(1/239)=:\pi_+.
\]
Exact rational comparisons give
$333/106<\pi_-<\pi_+<22/7$ and $\pi_->(4431/2500)^2$.
Use the rational endpoints
\[
 r_-=3162277660168379/10^{15},\qquad
 r_+=3162277660168380/10^{15},
\]
\[
 c_-=.4097321837023,\qquad c_+=.4097321837025.
\]
Squaring gives $r_-^2<10<r_+^2$; direct integer multiplication gives
\[
 c_-^2<\frac{19+6r_-}{72\pi_+}
 <\CE^2<\frac{19+6r_+}{72\pi_-}<c_+^2.
\]
This licenses the fine entry endpoints; the displayed coarse bounds for
$\CE$ follow since $819/2000<c_-<c_+<20487/50000<41/100$.
All denominators are positive and explicit. Finally $\pi>25/8$ gives the
coarse upper bounds, while $\pi<22/7$ and $(.39886)^2<7/44$ give the
lower bound for $\phi(0)$.
\end{proof}
